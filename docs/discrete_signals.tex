\chapter{Discrete signals}

\section{(Continuous) Signals}

An signal $s$ is the result of the measurement of some physical
event. For example, if $x$ represents a SPL (sound preasure level),
$x(t)$ is the SPL value at the instant of time $t$.

\section{Discrete signals}
\label{sec:sampling}
A discrete signal $\mathbf{x}$ is a sequence of samples $\mathbf{x}_n$,
\begin{equation}
  \mathbf{x} = \{\mathbf{x}_n\}_{n=0}^{N-1},
\end{equation}
where $N$ is the signal length\footnote{By physical reasons, we will
suppose that discrete signals are finite.}. Continuing with the SPL
example, each sample is ideally found with
\begin{equation}
  \mathbf{x}_n = x(t)\delta(t-nT),
\end{equation}
where $T$ is the sampling period, and
$\delta(t-nT)$ is the \emph{unit impulse function} defined by
\begin{equation}
\delta(t) =
\begin{cases}
\infty & \text{for } t = 0 \\
0 & \text{for } t \neq 0,
\end{cases}
\end{equation}
where
\begin{equation}
\int_{-\infty}^{\infty} \delta(t) \, dt = 1.
\end{equation}

Notice that, an impulse has the so-called shifting property with
respect to integration,
\begin{equation}
\int_{-\infty}^{\infty} x(t)\delta(t) \, dt = x(0),
\end{equation}
provided that $x(t)$ is continuous at $t=0$ \cite{gonzalez1992digital}.

\section{Energy of a discrete signal}
\label{sec:energy_signal}
%{{{

The energy of a discrete signal $\mathbf{x}$ is the sum of the square
of all its samples:
\begin{equation}
  E(\mathbf{x}) = \sum_{i=0}^{N-1}|\mathbf{x}_i|^2.
\end{equation}

Notice that, by definition, the energy of a finite-length discrete
signal is finite, but grows with $N$. Obviously, if $N$ where infinite,
the energy would be also infinite.

%}}}

\section{Power of a discrete signal}
\label{sec:power_signal}
%{{{

The power of a discrete signal $\mathbf{x}$ is
\begin{equation}
  P(\mathbf{x}) = \frac{1}{N}E(\mathbf{x}),
  \label{eq:power_discrete_signal}
\end{equation}
i.e., its average energy (average of the squared magnitude of the
signal values).

Compared to the energy of a signal, the use of the power of a signal
can be more convenient to avoid arithmetic overflow in computations
with long signals.

%}}}

\section{Fourier transform of a continuous signal}
\label{sec:Fourier_transform}
%{{{

The Fourier transform, usually denoted by $\mathcal{F}$, of a
continuous function $x(t)$ is
\begin{equation}
  \mathcal{F}(x) = X(w) = \int_{-\infty}^{\infty}x(t)e^{-jwt}dt
  \label{eq:FT}
\end{equation}
and its inverse, denoted by $\mathcal{F}^{-1}$ is
\begin{equation}
  x(t) = \mathcal{F}^{-1}(X(w)) = \int_{-\infty}^{\infty}X(w)e^{jtw}dw,
\end{equation}
where $w=2\pi f$ denotes angular frequency, $j=\sqrt{-1}$, and
$f$ is frequency.

Notice that, the Fourier transform of a signal $x$ is a complex signal
$X$, even if $x$ is real.

%}}}

\section{\glsentryfull{DFT}}
\label{sec:DFT}
%{{{

The \acrshort{DFT} of a (by definition, discrete finite) signal
$\mathbf{x}$ is determined by
\begin{equation}
  \mathcal{F}(\mathbf{x}) = \mathbf{X} = \{\mathbf{X}_f\}_{f=0}^{N-1} \quad \text{with} \quad \mathbf{X}_f=\sum_{n=0}^{N-1}\mathbf{x}_ne^{-2\pi jf\frac{n}{N}},
  \label{eq:DFT}
\end{equation}
where $f$ denotes (discrete) frequency bins. The inverse \gls{DFT} is defined by
\begin{equation}
  \mathbf{x} = \mathcal{F}^{-1}(\mathbf{X}) \quad \text{with} \quad \mathbf{x}_n = \frac{1}{N}\sum_{f=0}^{N-1}\mathbf{X}_fe^{2\pi jn\frac{f}{N}}.
\end{equation}

The \gls{DFT} can be interpreted as sampling the \gls{FTDF} (see
Appendix~\ref{sec:FTDF}) at $N$ evenly spaced points over $[-\pi,
  \pi)$. So, the \gls{DFT} is a discrete approximation of the
  \gls{FTDF} for finite-length signals. If $N$ were infinite, what
  happens is that the frequency bins would be infinitesimally narrow,
  and the Fourier transform of the discrete signal would be
  continuous.

Notice that there exists a faster version of the \gls{DFT} (with have
a complexity of $\mathcal{O}^2$ called \gls{FFT} with complexity
$\mathcal{O}\log_2\mathcal{O}$.

%}}}

\section{Parseval's theorem}
\label{sec:parseval}
%{{{

The Parseval's theorem for discrete signals states that, except for a
scale factor, the total energy of a signal in the signal domain (time,
for example) is equal to the total energy of its representation in the
frequency domain. Concretely, if $\mathbf{X}$ is the \gls{DFT} of the
discrete (and finte) signal $\mathbf{x}$, the Parseval’s theorem
states that
\begin{equation}
  E(\mathbf{x}) = \frac{1}{N}E(\mathbf{X}),
\end{equation}
where $E(\cdot)$ represents the energy operator (see
Section~\ref{sec:energy_signal}).

%}}}

\section{Fourier spectrum}
\label{sec:Fourier_spectrum}
%{{{

By definition, the Fourier
(or frequency) spectrum of $\mathbf{x}$ is the magnitude of
$\mathbf{X}$, that is
\begin{equation}
  |\mathbf{X}| = \sqrt{(\text{Re}(\mathbf{X}))^2+(\text{Im}(\mathbf{X}))^2}.
\end{equation}
As a complex function, we can also find the \emph{phase angle},
\emph{argument} or \emph{phase spectrum} using
\begin{equation}
  \arg({\mathbf{X}}) = \text{arctan}\left(\frac{\text{Re}(\mathbf{X})}{\text{Im}(\mathbf{X})}\right),
\end{equation}
where $\text{arctan}()$ must be computed using a four-quadrant
arctangent function.

%}}}

\section{Power spectrum}
\label{sec:power_spectrum}
%{{{

The \gls{PS} of finite-length discrete signal $\mathbf{x}$ shows how
much power (see Section~\ref{sec:power_signal}) a signal has at its
different frequency bins. Let $\mathbf{X}$ the Fourier transform of
$\mathbf{x}$. Then,
\begin{equation}
  \text{PS}(\mathbf{x}) = \{\text{PS}(\mathbf{x})_f\}_{f=0}^{N-1} \quad \text{where} \quad \text{PS}(\mathbf{x})_f = \frac{1}{N}|\mathbf{X}_f|^2,
\end{equation}
where $f$ denotes the frequency bin.

Notice that, for real-valued digital signals, the spectrum is
symmetric about the $N/2$-th bin (Nyquist frequency).

%}}}

\section{Convolution}
\label{sec:convolution}
%{{{

Typically represented with the symbol $\ast$, convolution is a
mathematical operation that combines two signals to produce a third
one. From a physical perspective, convolution models how a signal
is transformed by a system's behavior what is itself described by
another signal (usually, the response of the system to the an
impulse). An example of such system can be a filter.

For two signals $\mathbf{x}$ (the signal to be filtered) and
$\mathbf{h}$ (the filter weights), the convolution is defined as
\begin{equation}
(\mathbf{x}\ast\mathbf{h})_{n}=\sum_{k=0}^{K-1}\mathbf{x}_{n-k}\mathbf{h}_{k},
\label{eq:convolution}
\end{equation}
where $K$ is the number of weights, and $n$ is known as the \emph{lag}
(where the filter is applied in $\mathbf{x}$).

Notice that we could need to extend $\mathbf{x}$ to convolve it with
$\mathbf{h}$.

%}}}

\section{Transfer function of a filter}
\label{sec:TF_filter}

The \gls{TF} of a (\gls{LTI}) filter describes how the filter modifies
the amplitude and phase of each frequency component of the input
signal. In the analog world, it is defined as
\begin{equation}
  H(s) = \frac{Y(s)}{X(s)},
\end{equation}
where $Y(s)$ is the Laplace transform of the output, $X(s)$ is the
Laplace transform of the input, and $s$ represents a complex frequency
(a point in the complex $s$-plane). In the digital case, the
Laplace-domain is changed by the Z-domain ($z$-plane), resulting
(for the case of a \gls{IIR} filter)
\begin{equation}
  H(z) = \frac{Y(z)}{X(z)} = \frac{b_0 + b_1 z^{-1} + b_2 z^{-2} + \dots + b_M z^{-M}}{1 + a_1 z^{-1} + a_2 z^{-2} + \dots + a_N z^{-N}}
\end{equation}
where $z=e^{jw}$, $Y(z)$ is the Z-transform of the output, $X(z)$ is the
Z-transform of the input, $\{b_k\}$ are the feedforward weights,
and $\{a_k\}$ are the feedback weights. In the \gls{FIR} case,
all $a_k=0$ for $k\ge 1$, and therefore
\begin{equation}
  H(z) = b_0 + b_1 z^{-1} + b_2 z^{-2} + \dots + b_M z^{-M}.
\end{equation}

In both analog and digital cases, the evaluation of the transfer function on
the imaginary axis (unit circle), respectively, gives the frequency
response: $H(j\omega)$ (analog), $H(e^{j\omega})$ (digital).

\section{Fourier-based convolution (the convolution theorem)}
\label{sec:convolution_theorem}
%{{{

The convolution theorem states that the convolution of two signals in
the signal domain is equivalent to the (coefficient-wise)
multiplication of their Fourier transforms and then using the
inverse Fourier transform:
\begin{equation}
  \mathbf{x}\ast\mathbf{h} = \mathcal{F}^{-1}(\mathbf{X}\mathbf{H}).
\end{equation}

Conversely, the Fourier transform of the product of two signals is
the convolution of their individual Fourier transforms:
\begin{equation}
  \mathcal{F}(\mathbf{x}\mathbf{h}) = \mathbf{X}\ast\mathbf{H}.
\end{equation}

We can compute the convolution in the Fourier domain with a complexity
of $O(N\log N)$, using the following steps:
\begin{enumerate}
\item Take the FFT of $\mathbf{x}$ ($O(N\log N)$):
  \begin{equation}
    \mathbf{X} = \mathcal{F}(x).
  \end{equation}
\item Take the FFT of $\mathbf{y}$ ($O(N\log N)$):
  \begin{equation}
    \mathbf{Y} = \mathcal{F}(y).
  \end{equation}
\item Multiply (element-wise) $\mathbf{X}$ by
  $\mathbf{Y}$:
  \begin{equation}
    \mathcal{F}(\mathbf{x}\ast\mathbf{y})=\mathbf{X}\mathbf{Y}.
  \end{equation}
\item Take the inverse FFT of the result to get the convolution
  function ($O(N\log N)$):
  \begin{equation}
    \mathbf{x}\ast\mathbf{y} = \mathcal{F}^{-1}(\mathcal{F}(\mathbf{x}\ast\mathbf{y}))
  \end{equation}
\end{enumerate}

Notice that to perform a convolution in the Fourier domain, both
signals must have the same shape, which likely will force to zero-pad
$\mathbf{h}$. Therefore, when $K$ iw big, this can speed-up the
convolution if we use \gls{FFT} algorithms.

%}}}

\section{Cross-correlation}
\label{sec:cross-correlation}
%{{{

Essentially, \gls{CC}, usually represented by the symbol
$\circledast$, is a measurement of how much two or more variables are
related to one another. The \gls{CC} can be used to find the
similarity between two signals $\mathbf{x}$ and $\mathbf{h}$ (both
usually with the same shape) of one relative to the other, as a
function of a time lag $n\in\{n\}_{n=0}^{N-1}$. The \gls{CC} is
defined as
\begin{equation}
  (\mathbf{x}\circledast\mathbf{h})_n=\sum_{k=0}^{N-1}{\mathbf{x}}_{n+k} \mathbf{h}_n.
  \label{eq:cross-correlation}
\end{equation}

Notice that the \gls{CC} is identical to the convolution except that
one of the signals is \emph{flipped} ($\mathbf{x}$ in our case).

The \gls{CC} can be normalized between 0 and 1 (\gls{NCC}) using
\begin{equation}
  \text{NCC}(\mathbf{x},\mathbf{h})=\frac{{\mathbf{s}\circledast\mathbf{h}}}{\sqrt{\sum_n \mathbf{x}_n^2 \sum_n \mathbf{h}_n^2}}.
\end{equation}

%}}}

\section{\glsentryfull{CPSD}}
\label{sec:CPSD}
%{{{

The \acrshort{CPSD}, also known as cross-spectrum, analyzes the
relationship between two different signals in the Fourier
domain. Therefore, the \gls{CPSD} quantifies the degree to which two
signals are correlated or ``statistically connected'' at specific
frequencies (a high \gls{CPSD} value at a particular frequency
indicates a strong correlation between the two signals at that
frequency).

The \gls{CPSD} between two signals $\mathbf{x}$ and $\mathbf{y}$ is
the Fourier transform of the \gls{CC} between these two
signals:
\begin{equation}
  \text{CPSD}(\mathbf{x},\mathbf{y})=\mathcal{F}({\mathbf{x}\circledast\mathbf{y}}).
\end{equation}

\section{Fourier-based cross-correlation (the cross-correlation theorem)}
%{{{

Considering the convolution in the Fourier fomain (Section~\ref{sec:convolution_theorem}), i.e.,
\begin{equation}
  \mathbf{x}\circledast\mathbf{h} = \mathcal{F}^{-1}(\mathbf{X}\mathbf{H}^*),
  \label{eq:FFT_CC}
\end{equation}
where $\mathbf{H}^*$ is the complex conjugate of $\mathbf{H}$, we can
compute the cross-correlation in the Fourier domain with a complexity
of $O(N\log N)$, using the following steps:
\begin{enumerate}
\item Take the FFT of $\mathbf{x}$ ($O(N\log N)$):
  \begin{equation}
    \mathbf{X} = \mathcal{F}(x).
  \end{equation}
\item Take the FFT of $\mathbf{y}$ ($O(N\log N)$):
  \begin{equation}
    \mathbf{Y} = \mathcal{F}(y).
  \end{equation}
\item Multiply (element-wise) $\mathbf{X}$ by the complex conjugate of
  $\mathbf{Y}$ to get the CPSD (Cross-Power Spectral Density) ($O(N)$):
  \begin{equation}
    \text{CPSD}(\mathbf{x},\mathbf{y})=\mathcal{F}(\mathbf{x}\circledast\mathbf{y})=\mathbf{X}\mathbf{Y}^*
  \end{equation}
\item Take the inverse FFT of the result to get the cross-correlation
  function ($O(N\log N)$):
  \begin{equation}
    \mathbf{PCC} = \mathcal{F}^{-1}(\text{CPSD}(\mathbf{x},\mathbf{y}))
  \end{equation}
\end{enumerate}

\section{Wiener-Khinching theorem}
\label{sec:WKT}
%{{{

The Wiener-Khinching theorem states that the \gls{PSD} of a signal is
the Fourier transform of its autocorrelation function:
\begin{equation}
  \text{PSD}(\mathbf{x}) = \mathcal{F}(\text{AC}(\mathbf{x})).
\end{equation}
Therefore, taking the inverse Fourier transform on both sides, we
obtain that
\begin{equation}
  \mathcal{F}^{-1}(\text{PSD}(\mathbf{x})) = \text{AC}(\mathbf{x}),
\end{equation}
which matches the Eq.~\ref{eq:FFT_CC}.

\section{Fourier transform of a discrete signal}
\label{sec:FTDF}
%{{{

If the signal $\mathbf{s}$ is discrete, its Fourier transform, also
known as the \gls{DTFT}, is
\begin{equation}
  \mathcal{F}(\mathbf{s}) = S(e^{jw}) = \sum_{i=1}^{N}\mathbf{s}_ie^{-jwi}
\end{equation}
which is a continuous function of $w$ if $N$, and its inverse is
\begin{equation}
  \mathbf{s} = \mathcal{F}^{-1}(S(e^{jw})) = \frac{1}{2\pi}\int_{-\pi}^{\pi}S(e^{jw})e^{jwn}dw.
\end{equation}
The \acrshort{FTDF} is periodic with period $2\pi$.

%}}}

\section{Dot product of two discrete signals}
The dot product of two signal $\mathbf{A}$ and $\mathbf{B}$ is
calculated multiplying the corresponding elements of both signals and
summing the result:
\begin{equation}
  \mathbf{A}\cdot\mathbf{B} = \sum_{i=0}^{n-1}\mathbf{A}_i\mathbf{B_i}.
\end{equation}
